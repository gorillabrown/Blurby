def digest(data):

def ordinary(path, directory=False):
 path = pathlib.Path(path)
 for ancestor in reversed(path.parents):
  s = ancestor.lstat()
  require(stat.S_ISDIR(s.st_mode) and not (s.st_file_attributes & 0x400), 'ANCESTOR_NOT_ORDINARY')

 require(stat.S_ISDIR(s.st_mode) if directory else stat.S_ISREG(s.st_mode), 'WRONG_FILE_KIND')
 return s

def raw(path, cap):
 s = ordinary(path)
 require(s.st_size <= cap, 'FILE_BOUND',dict(path=str(path),observedSize=s.st_size,maximumBytes=cap))

  (e.st_dev,e.st_ino,e.st_size,e.st_mtime_ns), 'FILE_CHANGED_DURING_READ')
 return value
