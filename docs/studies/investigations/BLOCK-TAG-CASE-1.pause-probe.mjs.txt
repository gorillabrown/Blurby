// Scratch (BLOCK-TAG-CASE-1, supplementary to Done-when 4): where do the Flow/Focus engine's paragraph pauses fall?
// Compares the Flow surface's pause set (getParagraphBreaks) with the stamped block ends of the loaded sections.
// Route A: open Meditations, switch to Flow at the start. Route B: Page, TOC jump to INTRODUCTION, then Flow
// (word-probe.mjs's route). node pause-probe.mjs <sweep dir> <exe> <profile> <port> <out.json>
import fs from "node:fs";
import path from "node:path";
const [sweepDir, exe, profile, portArg, outJson] = process.argv.slice(2);
const { Session, blurbyPlan, home, runSteps, sleep, hashLive } = await import("file:///" + path.join(sweepDir, "sweep.mjs").replace(/\\/g, "/"));
const live = path.join(process.env.APPDATA, "blurby", "blurby-data");
const liveBefore = hashLive(live);
fs.mkdirSync(profile, { recursive: true });
const plan = blurbyPlan({});
const out = { exe, liveBefore, routes: [] };
const READ = `(() => {
  const host = document.getElementById("root") || document.body.firstElementChild; const ck = host && Object.keys(host).find((x) => x.startsWith("__reactContainer$"));
  const stack = ck ? [host[ck]] : []; let n = 0, rt = null;
  while (stack.length && !rt && n++ < 400000) { const f = stack.pop(); if (!f) continue; const p = f.memoizedProps;
    if (p && p.runtime && typeof p.runtime.getSnapshot === "function" && p.runtime.mode === "flow") rt = p.runtime;
    if (f.sibling) stack.push(f.sibling); if (f.child) stack.push(f.child); }
  if (!rt) return { err: "no flow runtime" };
  const bw = rt.document.bookWords; const pause = [...rt.surface.getParagraphBreaks()];
  const BT = new Set(["p","div","h1","h2","h3","h4","h5","h6","blockquote","li","td","section","article"]);
  const fv = document.querySelector("foliate-view"); const loaded = []; const ends = new Set();
  for (const { doc, index } of fv.renderer.getContents()) { loaded.push(index);
    const last = new Map(); for (const s of doc.querySelectorAll("[data-word-index]")) { let el = s.parentElement; while (el && !BT.has(el.localName.toLowerCase())) el = el.parentElement; if (el) last.set(el, Number(s.getAttribute("data-word-index"))); }
    for (const g of last.values()) ends.add(g); }
  const sec = (g) => bw.sections.find((x) => g >= x.startWordIdx && g < x.endWordIdx)?.sectionIndex;
  const onEnd = pause.filter((g) => ends.has(g)).length;
  return { playing: rt.getSnapshot().playing, cur: rt.getSnapshot().currentWordIndex ?? null, loadedSections: loaded, sectionStarts: loaded.map((i) => bw.sections.find((x) => x.sectionIndex === i)?.startWordIdx),
    pauseCount: pause.length, pauseOnStampedBlockEnd: onEnd, pauseSections: [...new Set(pause.map(sec))], stampedBlockEnds: ends.size,
    sample: pause.slice(0, 10).map((g) => ({ g, word: bw.words[g], next: bw.words[g + 1], isBlockEnd: ends.has(g) })) };
})()`;
const s = new Session({ exe, profile, port: Number(portArg), liveProfile: live });
await s.start(); console.log("preflight OK:", s.userData); out.userData = s.userData;
await plan.seed(s, { seed: [] });
for (const route of ["A", "B"]) {
  await home(s, plan);
  await runSteps(s, [{ click: '.doc-card[data-doc-id="sample-meditations"]', sleep: 300 }, { wait: ".reader-layout", waitMs: 45000, sleep: 2500 }]);
  if (route === "B") {
    await runSteps(s, [{ click: '[aria-label="Page mode"]', sleep: 2500 }, { click: ".rbb-chapter-name", sleep: 600 }]);
    await s.cdp.eval(`(() => { const o = [...document.querySelectorAll("li.rbb-chapter-option")].find((x) => x.textContent.trim() === "INTRODUCTION"); if (o) o.click(); return !!o; })()`); await sleep(3000);
  }
  await runSteps(s, [{ click: '[aria-label="Flow mode"]', sleep: 1500 }, { wait: ".rm-flow-root", waitMs: 45000, sleep: 5000 }]);
  let r = await s.cdp.eval(READ); if (r.playing) { await s.key("Space"); await sleep(1500); r = await s.cdp.eval(READ); }
  out.routes.push({ route, ...r }); console.log(route, JSON.stringify(r).slice(0, 900));
}
await s.stop();
out.liveAfter = hashLive(live); out.liveUnchanged = JSON.stringify(out.liveBefore) === JSON.stringify(out.liveAfter);
fs.writeFileSync(outJson, JSON.stringify(out, null, 2)); console.log("liveUnchanged", out.liveUnchanged); process.exit(0);
