class Cdp {
  constructor(url) { this.url = url; this.id = 0; this.pending = new Map(); }
  async open() {
    this.ws = new WebSocket(this.url);
    await new Promise((res, rej) => { this.ws.onopen = res; this.ws.onerror = () => rej(new Error("CDP websocket error")); });
    this.ws.onmessage = (m) => { const d = JSON.parse(m.data); if (d.id && this.pending.has(d.id)) { const p = this.pending.get(d.id); this.pending.delete(d.id); d.error ? p.reject(new Error(`${p.method}: ${d.error.message}`)) : p.resolve(d.result); } };
    this.ws.onclose = () => { for (const p of this.pending.values()) p.reject(new Error("CDP closed")); this.pending.clear(); this.closed = true; };
  }
  send(method, params = {}, timeoutMs = 15000) {
    if (this.closed) return Promise.reject(new Error("CDP closed"));
    const id = ++this.id; this.ws.send(JSON.stringify({ id, method, params }));
    return new Promise((resolve, reject) => { this.pending.set(id, { resolve, reject, method }); setTimeout(() => { if (this.pending.has(id)) { this.pending.delete(id); reject(new Error(`${method}: timeout`)); } }, timeoutMs); });
  }
  async eval(expr, timeoutMs = 10000) {
    const r = await this.send("Runtime.evaluate", { expression: expr, awaitPromise: true, returnByValue: true, timeout: timeoutMs }, timeoutMs + 2000);
    if (r.exceptionDetails) throw new Error(`eval: ${r.exceptionDetails.exception?.description || r.exceptionDetails.text}`);
    return r.result.value;
  }
  close() { try { this.ws.close(); } catch { /* gone */ } }
}

// One action: wrap, act, then the cursor poll (10 s) on the saved word, saves, library.json 4 s later, order.
async function action(name, act) {
  const wrap = await page.eval(WRAP); await ordTake();
  const s0 = (await saves()).length; const before = await probe(); const wasPlaying = await playing();
  const how = await act();
  let cursor = null, sp = null, newSaves = [];
  for (let i = 0; i < 40; i++) {
    await sleep(250); newSaves = (await saves()).slice(s0);
    const p = await probe(); const word = newSaves.at(-1)?.position ?? (p?.sectionIndex !== before?.sectionIndex ? p?.first?.wi : null);
    if (word != null) { sp = await span(word); if (/page-word--(flow-cursor|highlighted)/.test(sp?.cls ?? "")) { cursor = { word, atMs: (i + 1) * 250 }; break; } }
  }
  await sleep(1000);
  const after = await probe(); newSaves = (await saves()).slice(s0); await sleep(4000); const lib = readLib(); const ord = await ordTake();
  const lastGoTo = ord.filter((o) => o.k === "goTo").at(-1);
  const resolvedAt = lastGoTo ? ord.findIndex((o) => o.k === "resolved" && o.id === lastGoTo.id) : -1;
  const goToAt = lastGoTo ? ord.indexOf(lastGoTo) : -1;
  const relocates = ord.map((o, i) => ({ ...o, i })).filter((o) => o.k === "relocate" && o.i > goToAt);
  const resolvedIds = ord.filter((o) => o.k === "resolved").map((o) => o.id);
  const goToIds = ord.filter((o) => o.k === "goTo").map((o) => ({ id: o.id, target: o.target, resolved: resolvedIds.includes(o.id) }));
  const r = { how, wrap, goToIds, before, after, wasPlaying, playingAfter: await playing(), newSaves: newSaves.map((x) => ({ ...x, spine: spineStep(x.cfi) })),
    afterSpine: spineStep(after?.cfi), library: { ...lib, spine: spineStep(lib.cfi) }, order: ord,
    relocateBeforeResolve: resolvedAt < 0 ? null : relocates.some((o) => o.i < resolvedAt), relocateAfterResolve: resolvedAt < 0 ? null : relocates.some((o) => o.i > resolvedAt),
    span: sp, cursor, cursors: await page.eval(CURSORS).catch(() => null) };
  step(name, r); return r;
}
