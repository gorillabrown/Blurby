class Cdp {
  constructor(url) { this.url = url; this.id = 0; this.pending = new Map(); }
  async open() {
    this.ws = new WebSocket(this.url);
    await new Promise((res, rej) => { this.ws.onopen = res; this.ws.onerror = () => rej(new Error("CDP websocket error")); });
    this.ws.onmessage = (m) => { const d = JSON.parse(m.data); if (d.id && this.pending.has(d.id)) { const p = this.pending.get(d.id); this.pending.delete(d.id); d.error ? p.reject(new Error(`${p.method}: ${d.error.message}`)) : p.resolve(d.result); } };
    this.ws.onclose = () => { for (const p of this.pending.values()) p.reject(new Error("CDP closed")); this.pending.clear(); this.closed = true; };
  }
  send(method, params = {}, timeoutMs = 15000) {
    if (this.closed) return Promise.reject(new Error("CDP closed"));
    const id = ++this.id; this.ws.send(JSON.stringify({ id, method, params }));
    return new Promise((resolve, reject) => { this.pending.set(id, { resolve, reject, method }); setTimeout(() => { if (this.pending.has(id)) { this.pending.delete(id); reject(new Error(`${method}: timeout`)); } }, timeoutMs); });
  }
  async eval(expr, timeoutMs = 10000) {
    const r = await this.send("Runtime.evaluate", { expression: expr, awaitPromise: true, returnByValue: true, timeout: timeoutMs }, timeoutMs + 2000);
    if (r.exceptionDetails) throw new Error(`eval: ${r.exceptionDetails.exception?.description || r.exceptionDetails.text}`);
    return r.result.value;
  }
  close() { try { this.ws.close(); } catch { /* gone */ } }
}

let proc, exited, page;
const appLog = [];
async function finish(code) {
  try { page?.close(); proc?.kill(); } catch { /* gone */ }
  await Promise.race([exited, sleep(5000)]);
  rec.liveAfter = hashLive(); rec.liveUnchanged = JSON.stringify(rec.liveBefore) === JSON.stringify(rec.liveAfter);
  rec.appLogErrors = appLog.filter((l) => /error|unhandled|crash/i.test(l)).slice(-40);
  rec.finished = new Date().toISOString(); save();
  log("live unchanged:", rec.liveUnchanged); process.exit(code);
}
async function abort(why, code = 4) { step("ABORT", why); console.error("ABORT:", why); await finish(code); }

async function clickAt(b) { for (const type of ["mouseMoved", "mousePressed", "mouseReleased"]) await page.send("Input.dispatchMouseEvent", { type, x: b.x, y: b.y, button: "left", clickCount: 1 }); }

const KEYS = {
  Escape: { key: "Escape", code: "Escape", vk: 27 }, ArrowDown: { key: "ArrowDown", code: "ArrowDown", vk: 40 },
  ";": { key: ";", code: "Semicolon", vk: 186, text: ";" }, g: { key: "g", code: "KeyG", vk: 71, text: "g" },
  h: { key: "h", code: "KeyH", vk: 72, text: "h" }, l: { key: "l", code: "KeyL", vk: 76, text: "l" },

async function key(name) {
  const k = KEYS[name];
  await page.send("Input.dispatchKeyEvent", { type: k.text ? "keyDown" : "rawKeyDown", key: k.key, code: k.code, windowsVirtualKeyCode: k.vk, modifiers: k.mods || 0, text: k.text });
  await page.send("Input.dispatchKeyEvent", { type: "keyUp", key: k.key, code: k.code, windowsVirtualKeyCode: k.vk, modifiers: k.mods || 0 });
}

// BUG-215: no route in the app opens quick settings (Ctrl+Shift+, reaches the Ctrl+, branch). The probe opens it by
// setting the same activeOverlay state `;` sets: open highlights with `;`, then call that hook's own dispatcher.
const OPEN_QS = `(() => { const el = document.querySelector('.highlights-overlay'); if (!el) return 'no highlights overlay';
  const k = Object.keys(el).find((x) => x.startsWith('__reactFiber$')); let f = el[k];
  for (; f; f = f.return) { for (let h = f.memoizedState; h && typeof h === 'object' && 'next' in h; h = h.next) {
    if (h.memoizedState === 'highlights' && h.queue && typeof h.queue.dispatch === 'function') { h.queue.dispatch('quickSettings'); return 'dispatched'; } } }
  return 'no activeOverlay hook'; })()`;
